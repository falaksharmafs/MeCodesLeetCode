class Solution {
public:
    vector<int> findMissingElements(vector<int>& nums) {

        int minn = INT_MAX;
        int maxx = INT_MIN;

        unordered_set<int>seen;

        for(int x : nums){
            maxx = max(maxx,x);
            minn = min(minn,x);

            seen.insert(x);
        }

        vector<int>ans;

        for(int x = minn+1 ; x <maxx;x++){

            if(!seen.count(x)){
                ans.push_back(x);
            }

            

        }

        return ans;
        
    }
};
